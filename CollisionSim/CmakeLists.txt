cmake_minimum_required(VERSION 3.20)
project(CollisionSim LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(glfw3 CONFIG REQUIRED)
find_package(glad CONFIG REQUIRED)
find_package(glm CONFIG REQUIRED)
find_package(OpenGL REQUIRED)

add_executable(CollisionSim src/main.cpp)

target_link_libraries(CollisionSim PRIVATE glfw glad::glad glm::glm OpenGL::GL)

# Copy shaders next to the exe after build
add_custom_command(TARGET CollisionSim POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory
  ${CMAKE_SOURCE_DIR}/shaders $<TARGET_FILE_DIR:CollisionSim>/shaders)
